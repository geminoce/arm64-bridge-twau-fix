cmake_minimum_required(VERSION 3.22)
project(wolf_arm64_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DYNARMIC_FRONTENDS A32 CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(Boost_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/boost")
set(BOOST_ROOT "${Boost_INCLUDE_DIR}")
set(Boost_NO_SYSTEM_PATHS ON)
include_directories("${Boost_INCLUDE_DIR}")
add_compile_options(-fvisibility=hidden -fvisibility-inlines-hidden -ffunction-sections -fdata-sections)
add_link_options(-Wl,--gc-sections -Wl,--icf=all -Wl,--exclude-libs,ALL)
add_subdirectory(third_party/dynarmic dynarmic_build)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(GUEST_QSORT_HEADER "${CMAKE_CURRENT_BINARY_DIR}/guest_qsort_image.h")
add_custom_command(OUTPUT "${GUEST_QSORT_HEADER}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build_guest_qsort.py"
        --ndk "${CMAKE_ANDROID_NDK}" --source "${CMAKE_CURRENT_SOURCE_DIR}/src/guest_qsort.c"
        --output "${GUEST_QSORT_HEADER}" --work-dir "${CMAKE_CURRENT_BINARY_DIR}/guest_qsort"
    DEPENDS build_guest_qsort.py src/guest_qsort.c VERBATIM)
add_custom_target(guest_qsort_image DEPENDS "${GUEST_QSORT_HEADER}")
set(WOLF_CORE_SOURCES
    src/elf_loader.cpp src/dynarmic_host.cpp src/thunk_manager.cpp
    src/thunks_gl.cpp src/thunks_gl_gles2_core.cpp src/thunks_android.cpp
    src/thunks_libc.cpp src/thunks_opensles.cpp src/jni_fake_env.cpp)
add_library(wolf_core OBJECT ${WOLF_CORE_SOURCES})
add_dependencies(wolf_core guest_qsort_image)
target_include_directories(wolf_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(wolf_core PUBLIC ANDROID __ANDROID__ ARM32_BRIDGE_ENABLE_LOGGING=1)
option(WOLF_FMOD_DIAGNOSTIC "Observe four original FMOD call boundaries with bounded state capture" OFF)
if(WOLF_FMOD_DIAGNOSTIC)
    target_compile_definitions(wolf_core PUBLIC WOLF_FMOD_DIAGNOSTIC=1)
endif()
if(WOLF_FMOD_DIAGNOSTIC)
    target_link_libraries(wolf_core PRIVATE merry::mcl)
endif()
target_link_libraries(wolf_core PUBLIC dynarmic GLESv3 GLESv2 GLESv1_CM EGL OpenSLES log android m z)
target_link_options(wolf_core INTERFACE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384 -Wl,-z,relro -Wl,-z,now)
add_library(main SHARED src/jni_bridge.cpp)
target_link_libraries(main PRIVATE wolf_core)
add_executable(wolf_fastmem_fpsr_contract_test EXCLUDE_FROM_ALL tests/fastmem_fpsr_contract_test.cpp)
target_link_libraries(wolf_fastmem_fpsr_contract_test PRIVATE dynarmic merry::mcl)
target_compile_options(wolf_fastmem_fpsr_contract_test PRIVATE -Wall -Wextra -Werror)
add_executable(wolf_display_split_contract_test tests/display_split_contract_test.cpp)
target_link_libraries(wolf_display_split_contract_test PRIVATE wolf_core)
target_compile_options(wolf_display_split_contract_test PRIVATE -Wall -Wextra -Werror)
add_executable(wolf_renderer_state_contract_test EXCLUDE_FROM_ALL tests/renderer_state_contract_test.cpp)
target_link_libraries(wolf_renderer_state_contract_test PRIVATE wolf_core)
target_compile_options(wolf_renderer_state_contract_test PRIVATE -Wall -Wextra -Werror)
add_executable(wolf_shader_compat_test EXCLUDE_FROM_ALL tests/wolf_shader_compat_test.cpp)
target_include_directories(wolf_shader_compat_test PRIVATE src)
target_link_libraries(wolf_shader_compat_test PRIVATE EGL GLESv3)
target_compile_options(wolf_shader_compat_test PRIVATE -Wall -Wextra -Werror)
option(WOLF_JNI_CONTRACT_CHECKS "Run bounded ART contracts before guest bootstrap" OFF)
if(WOLF_JNI_CONTRACT_CHECKS)
    add_library(wolf_fakejni_art_checks OBJECT tests/fakejni_art_checks.cpp)
    target_include_directories(wolf_fakejni_art_checks PRIVATE src)
    target_compile_options(wolf_fakejni_art_checks PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(wolf_fakejni_art_checks PRIVATE dynarmic)
    target_sources(main PRIVATE $<TARGET_OBJECTS:wolf_fakejni_art_checks>)
    target_compile_definitions(main PRIVATE WOLF_JNI_CONTRACT_CHECKS=1)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/loader_libc_contract_test.cpp")
    add_executable(wolf_loader_libc_contract_test tests/loader_libc_contract_test.cpp)
    target_link_libraries(wolf_loader_libc_contract_test PRIVATE wolf_core)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/abi_boundary.cpp")
    add_library(wolf_jni_abi_compile OBJECT tests/abi_boundary.cpp)
    target_include_directories(wolf_jni_abi_compile PRIVATE src)
    target_compile_options(wolf_jni_abi_compile PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(wolf_jni_abi_compile PRIVATE dynarmic)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/graphics_contract_test.cpp")
    add_executable(wolf_graphics_contract_test tests/graphics_contract_main.cpp tests/graphics_contract_test.cpp)
    target_link_libraries(wolf_graphics_contract_test PRIVATE wolf_core)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/fmod_observer_contract_test.cpp")
    add_executable(wolf_fmod_observer_contract_test tests/fmod_observer_contract_test.cpp)
    target_link_libraries(wolf_fmod_observer_contract_test PRIVATE wolf_core)
endif()
